        # Check formats
        for img in images:
            if not self.is_supported_format(img):
                results["errors"].append(f"Unsupported format: {img.format}")
        # Check modality
        modalities = [self.detect_modality(img) for img in images]
        # → "optical" | "sar" | "multispectral"
        # Determine configuration
        if len(images) == 1:
            results["image_config"] = "SINGLE"
        elif len(images) == 2:

                results["image_config"] = "CROSS_MODAL"

# BigEarthNet Preprocessing Pipeline
class BigEarthNetPreprocessor:
    def prepare_training_data(self):
        # Load Sentinel-2 patches
        s2_patches = self.load_s2_images()  # 12 spectral bands
        # Load Sentinel-1 patches  
        s1_patches = self.load_s1_images()  # VV, VH polarizations
        # Generate text descriptions from labels
        labels = self.load_labels()
        text_descriptions = self.generate_text(labels)
        # Create image-text pairs for CLIP fine-tuning
        training_pairs = [
            {
            }
                "image": s2_patches[i],
                "text": text_descriptions[i],
                "sar": s1_patches[i]
            for i in range(len(s2_patches))
        ]
        return training_pairs
    def generate_text(self, labels):
        # Convert label list to natural language
        # ["Continuous urban fabric"] → "Dense urban area with continuous built
up fabric"
        templates = {
            "Continuous urban fabric": 
                "Dense urban area with continuous residential buildings",
            "Arable land": 
                "Open agricultural land used for crop cultivation",
            "Water bodies": 
                "Natural or artificial water body including lakes and rivers",
            # ... all 43 classes
        }
        return [templates.get(l, l) for l in labels]
